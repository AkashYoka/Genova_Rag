def _leaf_headers(header_rows: Sequence[Sequence[str]]) -> List[str]:
    if not header_rows:
        return []
    width = max(len(r) for r in header_rows)
    if width == 0:
        return []
    padded = [list(r) + [""] * (width - len(r)) for r in header_rows]
    if len(padded) == 1:
        return [normalize_ws(c) for c in padded[0]]
    leaves: List[str] = []
    for c in range(width):
        parts: List[str] = []
        for r in padded:
            val = normalize_ws(r[c])
            if val and (not parts or parts[-1].lower() != val.lower()):
                parts.append(val)
        if not parts:
            leaves.append(f"Column {c + 1}")
            continue
        composed = _compose_weight_dose_leaf(parts)
        if composed and (
            _WEIGHT_RANGE_RE.search(composed)
            or "dose" in composed.lower()
            or "weight" in composed.lower()
        ):
            leaves.append(composed)
            continue
        child = parts[-1]
        if len(parts) >= 2 and re.search(r"\d", child) and not re.search(r"\d", parts[0]):
            # Prefer specific trailing band; still attach Dose/(mg) parent if present.
            dose_bits = [p for p in parts[:-1] if _DOSE_UNIT_RE.match(p) or p.lower() == "kg"]
            if dose_bits:
                leaves.append(_compose_weight_dose_leaf(parts))
            else:
                leaves.append(child)
        elif len(parts) > 1:
            leaves.append(normalize_ws(" ".join(parts)))
        else:
            leaves.append(child)
    return leaves
